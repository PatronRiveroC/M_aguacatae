load_predictors <- function(path){
  files <- list.files(path, pattern = "\\.tif$", full.names = TRUE)
  terra::rast(files)
}

read_occurrences <- function(file){
  line <- readLines(file, n = 1)
    if(grepl("\t",line)){
      read.csv(file, sep = "\t", stringsAsFactors = FALSE)
    }else{
      read.csv(file, sep = ",", stringsAsFactors = FALSE)
    }
}

standardize_occurrence_columns <- function(data, species_name = "NULL"){
  if(all(c("species", "decimalLongitude", "decimalLatitude") %in% names(data))){
    data <- data[,c("species", "decimalLongitude", "decimalLatitude")]
  } else if(all(c("scientificname", "decimallongitude", "decimallatitude") %in% names(data))){
      data <- data[,c("scientificname", "decimallongitude", "decimallatitude")]
  } else if(all(c("Long", "Lat") %in% names(data))){
      data <- data[,c("Long", "Lat")]
      data$species <- species_name
      data <- data[,c("species", "Long", "Lat")]
  } else if(all(c("longitude", "latitude") %in% names(data))){
      data <- data[,c("longitude", "latitude")]
      data$species <- species_name
      data <- data[,c("species", "longitude", "latitude")]
  } else {
    stop("No compatible coordinate columns were found.")
  }
  colnames(data) <- c("Species", "Long", "Lat")
  return(data)
}

clean_occurrences <- function(data){
  data <- na.omit(data)
  data <- data[!duplicated(data[,c("Long","Lat")]),]
  data <- data %>%
  filter(Long >= -180, Long <= 180, Lat >= -90, Lat <= 9)
  data
}

extract_environment <- function(occurrences, predictors){
  env <- terra::extract(predictors, occurrences[,c("Long","Lat")], ID = FALSE)
  na.omit(cbind(occurrences, env))
}

load_predictors <- function(path){
  files <- list.files(path, pattern = "\\.tif$", full.names = TRUE)
  terra::rast(files)
}

select_variables_vif <- function(occ, predictors, threshold = 10){
  occ_sf <- terra::vect(occ, geom = c("Long","Lat"), crs = terra::crs(predictors))
  env <- terra::extract(predictors, occ_sf, ID = FALSE)
  env <- na.omit(env)
  vif <- usdm::vifstep(env, th = threshold)
  list(vif = vif, variables = vif@results$Variables)
}

tru <- raster(tru)
